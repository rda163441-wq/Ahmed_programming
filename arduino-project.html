What it does

One controller, several useful responses

The sketch reads digital sensor inputs, shows their status on a 16×2 display, and drives lights, a buzzer, and a servo.

⌁

Fire & gas alert

A low signal from either sensor switches on the warning LED and requests a buzzer alert.

☼

Automatic lighting

The LDR module's digital output switches two lights for the night condition.

▤

Live status display

The I2C LCD displays fire, gas, day/night, and door state labels.

↗

Servo door action

An active-low IR input turns the servo to 90° briefly, then returns it to 0°.

Control flow

How the sketch behaves

01 / READ

Sample sensor inputs

Each loop reads the flame, gas, LDR, IR, and button as digital values and logs them over Serial at 9600 baud.

02 / DECIDE

Update status & alerts

Low flame or gas readings trigger the warning LED; the later button check can override the buzzer. LDR and IR values select lighting and door states.

03 / ACT

Drive outputs

The LEDs, buzzer, and servo respond to the readings. The servo waits three seconds before returning to its closed position.

Source code

The Arduino sketch

Uses the Wire, LiquidCrystal_I2C, and Servo libraries.

Show the full source code 
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <Servo.h>

Servo servo;
LiquidCrystal_I2C lcd(0x27, 16, 2);

int flame = 13;
int ir = 12;
int gas = 2;
int LDR = 11;
int led_1 = 10;
int led_2 = 9;
int led_3 = 8;
int led_4 = 7;
int buzzer = 6;
int pb = 4;

int value_flame;
int value_gas;
int value_LDR;
int value_ir;
int value_pb;

void setup() {
  Serial.begin(9600);
  lcd.init();
  lcd.backlight();

  lcd.setCursor(0, 0);
  lcd.print(" WELCOME ");
  lcd.setCursor(0, 1);
  lcd.print(" SMART HOME");
  delay(3000);
  lcd.clear();

  pinMode(flame, INPUT);
  pinMode(gas, INPUT);
  pinMode(LDR, INPUT);
  pinMode(ir, INPUT);
  pinMode(led_1, OUTPUT);
  pinMode(led_2, OUTPUT);
  pinMode(led_3, OUTPUT);
  pinMode(led_4, OUTPUT);
  pinMode(buzzer, OUTPUT);
  pinMode(pb, INPUT);

  servo.attach(3);
}

void loop() {
  value_flame = digitalRead(flame);
  value_ir = digitalRead(ir);
  value_LDR = digitalRead(LDR);
  value_gas = digitalRead(gas);
  value_pb = digitalRead(pb);

  Serial.print("value_flame = ");
  Serial.print(value_flame);
  Serial.print(" value_ir = ");
  Serial.print(value_ir);
  Serial.print(" value_LDR = ");
  Serial.print(value_LDR);
  Serial.print(" value_gas = ");
  Serial.print(value_gas);
  Serial.print(" value_pb = ");
  Serial.println(value_pb);
  delay(200);

  if (value_flame == 0 || value_gas == 0) {
    digitalWrite(buzzer, HIGH);
    digitalWrite(led_3, HIGH);
  } else {
    digitalWrite(buzzer, LOW);
    digitalWrite(led_3, LOW);
  }

  if (value_flame == 0) {
    lcd.setCursor(0, 0);
    lcd.print("Fire ");
  } else {
    lcd.setCursor(0, 0);
    lcd.print("No Fire");
  }

  if (value_gas == 0) {
    lcd.setCursor(0, 1);
    lcd.print("Gas ");
  } else {
    lcd.setCursor(0, 1);
    lcd.print("No Gas");
  }

  if (value_LDR == 1) {
    digitalWrite(led_1, HIGH);
    digitalWrite(led_2, HIGH);
    lcd.setCursor(8, 1);
    lcd.print("NIGHT");
  } else {
    digitalWrite(led_1, LOW);
    digitalWrite(led_2, LOW);
    lcd.setCursor(8, 1);
    lcd.print("DAY ");
  }

  if (value_pb == 1) {
    digitalWrite(buzzer, HIGH);
  } else {
    digitalWrite(buzzer, LOW);
  }

  if (value_ir == 0) {
    lcd.setCursor(8, 0);
    lcd.print("OPEN ");
    digitalWrite(led_4, HIGH);
    servo.write(90);
    delay(3000);
    servo.write(0);
  } else {
    lcd.setCursor(8, 0);
    lcd.print("CLOSE ");
    digitalWrite(led_4, LOW);
    servo.write(0);
  }
}

Prototype safety note

This is an educational prototype, not a certified fire, gas, or security alarm. Do not rely on it for life safety or real home protection. In the supplied sketch, the button check runs after the sensor alert and may switch the buzzer off. Sensor modules, input polarity, button wiring, and power requirements must be verified for the actual hardware.
